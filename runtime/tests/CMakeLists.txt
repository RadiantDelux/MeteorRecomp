cmake_minimum_required(VERSION 3.22)
project(WiiCompiledRuntimeContracts LANGUAGES C CXX)

# This entry point deliberately has no renderer/download/generated-code graph.
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(Clang|AppleClang)$" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Runtime contracts require 64-bit Clang")
endif()
get_filename_component(MKW_RUNTIME_DIR "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(MKW_PLATFORM_SOURCE "${MKW_RUNTIME_DIR}/src/platform/host_platform.cpp")
set(MKW_TRANSLATED_PPC_FP_OPTIONS -fno-fast-math -ffp-contract=off)
if(WIN32)
    if(NOT MINGW)
        message(FATAL_ERROR "Windows runtime contracts require LLVM-MinGW")
    endif()
    set(MKW_PLATFORM_WINDOWS TRUE)
elseif(APPLE)
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|ARM64)$")
        message(FATAL_ERROR "macOS contracts currently support arm64")
    endif()
    set(MKW_PLATFORM_MACOS TRUE)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(MKW_PLATFORM_LINUX TRUE)
    add_library(mkw_libco STATIC "${MKW_RUNTIME_DIR}/third_party/libco/libco.c")
    add_library(mkw::libco ALIAS mkw_libco)
    target_include_directories(mkw_libco PUBLIC "${MKW_RUNTIME_DIR}/third_party/libco")
else()
    message(FATAL_ERROR "Unsupported contract-test host")
endif()
include("${MKW_RUNTIME_DIR}/cmake/RuntimeTests.cmake")
